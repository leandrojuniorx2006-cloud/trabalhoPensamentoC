#include <stdio.h>

int main() {
    int numero, soma = 0, contador = 0;

    printf("Digite um numero (0 para encerrar): ");
    scanf("%d", &numero);

    while (numero != 0) {
        soma = soma + numero;
        contador = contador + 1;

        printf("Digite um numero (0 para encerrar): ");
        scanf("%d", &numero);
    }

    printf("Total de numeros: %d\n", contador);
    printf("Soma: %d\n", soma);

    return 0;
}

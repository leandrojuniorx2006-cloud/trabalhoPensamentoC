#include <stdio.h>

int main() {
    float nota;

    printf("Digite a nota (0 a 10): ");
    scanf("%f", &nota);

    while (nota < 0.0f || nota > 10.0f) {
        printf("Nota invalida! Digite um valor entre 0 e 10: ");
        scanf("%f", &nota);
    }

    printf("Nota aceita: %.1f\n", nota);

    return 0;
}

#include <stdio.h>

int main() {
    int opcao;
    float n1, n2;

    printf("=== MENU ===\n");
    printf("1 - Soma\n");
    printf("2 - Multiplicacao\n");
    printf("0 - Sair\n");
    printf("Escolha: ");
    scanf("%d", &opcao);

    while (opcao != 0) {
        if (opcao == 1) {
            printf("Digite dois numeros: ");
            scanf("%f %f", &n1, &n2);
            printf("Soma = %.2f\n", n1 + n2);
        } else if (opcao == 2) {
            printf("Digite dois numeros: ");
            scanf("%f %f", &n1, &n2);
            printf("Multiplicacao = %.2f\n", n1 * n2);
        } else {
            printf("Opcao invalida!\n");
        }

        printf("\n=== MENU ===\n");
        printf("1 - Soma\n");
        printf("2 - Multiplicacao\n");
        printf("0 - Sair\n");
        printf("Escolha: ");
        scanf("%d", &opcao);
    }

    printf("Sistema encerrado.\n");

    return 0;
}

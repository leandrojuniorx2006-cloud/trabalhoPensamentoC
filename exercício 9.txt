#include <stdio.h>
 
int main() {
    int n, a = 0, b = 1, c;
 
    printf("Quantos termos você quer ver? ");
    scanf("%d", &n);
 
    for (int i = 0; i < n; i++) {
        if (i == 0) {
            printf("%d", a);
        } else if (i == 1) {
            printf(", %d", b);
        } else {
            c = a + b;
            a = b;
            b = c;
            printf(", %d", b);
        }
    }
 
    printf("\n");
 
    return 0;
}
